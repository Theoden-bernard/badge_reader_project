\section{Apprentissages}
Ce projet m’a permis d’acquérir des compétences clés.

\begin{for_you}{Apprentissages}
    \textit{Apprentissages clés} :
    \begin{itemize}
        \item \textbf{Elixir/Phoenix} :
              \begin{itemize}
                  \item Maîtrise des \textbf{Contexts} pour organiser la logique métier.
                  \item Utilisation avancée d’\textbf{Ecto} (requêtes, transactions).
                  \item Implémentation des \textbf{plugs} pour l’authentification.
              \end{itemize}
        \item \textbf{DevOps} :
              \begin{itemize}
                  \item Configuration d'une \textbf{CI/CD}.
                  \item Automatisation des tests avec \textbf{GitHub Actions}.
              \end{itemize}
        \item \textbf{Sécurité} :
              \begin{itemize}
                  \item Implémentation du \textbf{RGPD} (droits utilisateurs, chiffrement).
                  \item Protection contre les \textbf{OWASP Top 10}.
                  \item Gestion de \textbf{sessions sécurisées}.
                  \newline
              \end{itemize}
    \end{itemize}

    \textit{Exemple d’amélioration de code} :
    \begin{lstlisting}
    # AVANT : Logique métier dans le contrôleur
    def create(conn, %{"badge" => badge_params}) do
      # Validation, création tout en une fonction !
      # ...
    end

    # APRÈS : Logique décomposée en Contexts
    def create(conn, %{"badge" => badge_params}) do
      case Badges.create_badge(badge_params) do
        {:ok, badge} ->
          conn
          |> put_flash(:info, "Badge créé")
          |> redirect(to: ~p"/badges/#{badge}")

        {:error, changeset} ->
          conn
          |> put_flash(:error, "Erreur")
          |> render(:new, changeset: changeset)
      end
    end

    \end{lstlisting}
\end{for_you}
