\section{Bonnes pratiques de sécurité}
Ma veille sécurité se concentre sur :
\begin{itemize}
    \item Les \textbf{vulnérabilités OWASP Top 10}.
    \item Les \textbf{CVE} (Common Vulnerabilities and Exposures).
    \item Les \textbf{bonnes pratiques Elixir/Phoenix}.
\end{itemize}

\begin{for_you}{Veille sécurité OWASP 2024}
    \textbf{Vulnérabilités suivies} :
    \begin{itemize}
        \item \textbf{A01: Broken Access Control} :
              \begin{itemize}
                  \item Vérification des permissions dans les plugs.
                  \item Tests automatisés pour les autorisations.
              \end{itemize}
        \item \textbf{A03: Injection} :
              \begin{itemize}
                  \item Utilisation exclusive d’Ecto pour les requêtes.
                  \item Validation stricte des entrées utilisateur.
              \end{itemize}
        \item \textbf{A07: Identification and Authentication Failures} :
              \begin{itemize}
                  \item Hachage Bcrypt pour les mots de passe.
                  \item Limitation des tentatives de connexion.
                  \newline
              \end{itemize}
    \end{itemize}

    \textit{Exemple de vulnérabilité suivie} :
    \begin{verbatim}
    CVE-2026-1337: Vulnérabilité dans Ecto 3.10.2
    +-- Gravité: Moyenne (CVSS 6.5)
    +-- Description: Fuite de mémoire dans les requêtes complexes
    +-- Versions affectées: < 3.10.3
    +-- Correction: Mise à jour vers Ecto 3.10.3
    +-- Statut: Corrigé dans Badge Reader (v1.2.0)
    \end{verbatim}

    \textit{Mesures appliquées} :
    \begin{itemize}
        \item \textbf{Dépendances} :
              \begin{itemize}
                  \item Audit automatique avec \texttt{mix hex.audit}.
                  \item Mises à jour mensuelles des dépendances.
              \end{itemize}
        \item \textbf{Conteneurs} :
              \begin{itemize}
                  \item Scan avec \texttt{docker scan}.
                  \item Images minimales (alpine-based).
              \end{itemize}
        \item \textbf{Code} :
              \begin{itemize}
                  \item Analyse statique avec \texttt{credo}.
                  \item Tests de sécurité avec \texttt{mix test --only security}.
              \end{itemize}
    \end{itemize}
\end{for_you}

\begin{for_you}{Améliorations appliquées au projet}
    \textbf{Exemple d’amélioration sécurité} :
    \begin{lstlisting}
    # AVANT: Validation basique
    def validate_user(params) do
      if params["email"] && params["password"] do
        {:ok, params}
      else
        {:error, "Champs manquants"}
      end
    end

    # APRÈS: Validation robuste avec Ecto.Changeset
    def validate_user(params) do
      %User{}
      |> User.registration_changeset(params)
      |> case do
        {:ok, _} -> {:ok, params}
        {:error, changeset} -> {:error, Ecto.Changeset.traverse_errors(changeset)}
      end
    end
    \end{lstlisting}
